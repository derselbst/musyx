SND_EMITTER

typedef struct SND_EMITTER {
	struct SND_EMITTER	*next;
	struct SND_EMITTER	*prev;
	SND_ROOM		*room;
	
	SND_PARAMETER_INFO	*paraInfo;
	
	u32			flags;
	SND_FVECTOR		pos;
	SND_FVECTOR		dir;		// Speed in units/second
	f32			maxDis;
	f32			maxVol;
	f32			minVol;
	f32			volPush;	// 0.0 = linear -> 1.0 = square
	SND_VOICEID		vid;
	u32			group;		// Group ID (by default FXID | 0x80000000) used to do volume priorities for continous emitters
	
	SND_FXID		fxid;
	u8			studio;
	u16			VolLevelCnt;	// Used during continous emitter allocation process
	f32			fade;		// Used to fade-in of continous emitters
				
			} SND_EMITTER;
