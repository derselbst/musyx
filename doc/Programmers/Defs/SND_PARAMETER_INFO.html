SND_PARAMETER_INFO


typedef struct SND_PARAMETER_INFO {
	u8		numPara;	// How many MIDI controller values (ID,value - value may be 8-bit or 16-bit!)
	SND_PARAMETER	*paraArray;	// Parameter data...
				
	} SND_PARAMETER_INFO;
numPara defines how many MIDI controllers are to be set.
paraArray has to point to a list of these parameters. For each of these controllers, the controller ID and the value has to be specified. The value may be 7-bit or 14-bit depending on the controller ID.
